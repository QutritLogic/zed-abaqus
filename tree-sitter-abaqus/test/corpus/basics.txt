==================
Nested sections, case-insensitive, no final newline
==================

*ASSEMBLY, name=A
*Instance, name=I, part=P
1., 0., 0.
*end instance
*End Assembly
*Step, name=S
*Static
*End Step

---

(source_file
  (assembly
    start: (keyword_block
      header: (keyword_line
        name: (keyword_name)
        parameter: (parameter name: (parameter_name) value: (value))))
    (instance
      start: (keyword_block
        header: (keyword_line
          name: (keyword_name)
          parameter: (parameter name: (parameter_name) value: (value))
          parameter: (parameter name: (parameter_name) value: (value)))
        (data_line (number) (number) (number)))
      end: (keyword_line name: (keyword_name)))
    end: (keyword_line name: (keyword_name)))
  (step
    start: (keyword_block
      header: (keyword_line
        name: (keyword_name)
        parameter: (parameter name: (parameter_name) value: (value))))
    (keyword_block header: (keyword_line name: (keyword_name)))
    end: (keyword_line name: (keyword_name))))
